class Solution {
public:
    void rotate(vector<int>& nums, int k) {
        int len = nums.size()-1;
        int l=0, r=len;

        k = k%(len+1);

        while(l<r){
            swap(nums[l], nums[r]);
            l++;
            r--;
        }
        l=0, r=k-1;
        while(l<r){
            swap(nums[l], nums[r]);
            l++;
            r--;
        }
        l=k, r=len;
        while(l<r){
            swap(nums[l], nums[r]);
            l++;
            r--;
        }
    }
};
