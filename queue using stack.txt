class MyQueue {
public:
    stack<int> s1;
    stack<int> s2;
    MyQueue() {}

    void push(int x) { s1.push(x); }

    int pop() {
        if (s2.empty()) {
            while (!s1.empty()) {
                int n = s1.top();
                s1.pop();
                s2.push(n);
            }

            int d = s2.top();
            s2.pop();
            return d;
        }
        int d = s2.top();
        s2.pop();
        return d;
    }

    int peek() {
        if (s2.empty()) {
            while (!s1.empty()) {
                int n = s1.top();
                s1.pop();
                s2.push(n);
            }

            return s2.top();
        }
        return s2.top();
    }

    bool empty() {
        if (s1.empty() && s2.empty())
            return true;
        return false;
    }
};

 * int param_3 = obj->peek();
